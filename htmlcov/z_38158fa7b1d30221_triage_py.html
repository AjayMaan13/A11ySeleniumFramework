17def collect_violations(targets=None, browser="chrome", headless=True): 

18 """ 

19 Scan each configured target page and return a flat list of raw 

20 axe-core violations, each tagged with the page_url it came from - 

21 the input shape group_by_root_cause() expects. 

22 

23 Args: 

24 targets: list of {'url', 'expected_wcag_level'} dicts, e.g. from 

25 data/targets.json via src.utils.targets.TARGETS. Defaults to 

26 that list if not given. 

27 browser: browser to scan with ("chrome" or "firefox") 

28 headless: run the browser headless 

29 

30 Returns: 

31 List of axe-core violation dicts, each with a 'page_url' key added. 

32 """ 

33 targets = DEFAULT_TARGETS if targets is None else targets 

34 

35 driver = setup_driver(browser, headless) 

36 violations = [] 

37 try: 

38 for target in targets: 

39 url = target["url"] 

40 expected_level = target.get("expected_wcag_level", "AA") 

41 

42 scanner = AccessibilityScanner(driver) 

43 page = BasePage(driver) 

44 page.open(url) 

45 time.sleep(1) 

46 scanner.inject_axe() 

47 

48 results = scanner.run_custom_scan( 

49 options={ 

50 "runOnly": { 

51 "type": "tag", 

52 "values": WCAG_LEVEL_TAGS.get(expected_level, WCAG_LEVEL_TAGS["AA"]), 

53 } 

54 } 

55 ) 

56 

57 for violation in scanner.get_violations(results): 

58 violation["page_url"] = url 

59 violations.append(violation) 

60 finally: 

61 teardown_driver(driver) 

62 

63 return violations

66def run_triage(violations: list, llm, dedup, ticketer) -> list: 

67 """ 

68 Group, prioritize, and file tickets for the violations that warrant one. 

69 

70 Args: 

71 violations: raw axe-core violations (see collect_violations) 

72 llm: an object with describe_group(group) -> {'summary', 'fix', 

73 'ticket_worthy'} (src.agent.llm_client.LLMClient) 

74 dedup: an object with already_filed(fp) -> bool and 

75 record(fp, issue_number) (src.agent.ticket_dedup.DedupStore) 

76 ticketer: an object with create(group, description) -> issue id 

77 (src.integrations.github_issues.GitHubIssueCreator) 

78 

79 Returns: 

80 List of issue numbers filed on this run. 

81 """ 

82 groups = prioritize(group_by_root_cause(violations)) 

83 filed = [] 

84 

85 for group in groups: 

86 fp = fingerprint(group) 

87 if dedup.already_filed(fp): 

88 continue # decision: skip, already tracked 

89 

90 description = llm.describe_group(group) 

91 if not description["ticket_worthy"]: 

92 continue # decision: not worth a ticket 

93 

94 issue_number = ticketer.create(group, description) 

95 dedup.record(fp, issue_number) 

96 filed.append(issue_number) 

97 

98 return filed
