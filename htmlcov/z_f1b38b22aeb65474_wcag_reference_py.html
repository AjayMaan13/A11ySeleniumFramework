55# Function to get explanation of a WCAG term 

56def get_wcag_explanation(term): 

57 """ 

58 Get explanation for a WCAG term or guideline 

59  

60 Args: 

61 term: WCAG term or guideline number 

62  

63 Returns: 

64 Explanation string or None if not found 

65 """ 

66 return WCAG_GLOSSARY.get(term) 

67 

68# Function to get all terms matching a category 

69def get_terms_by_category(category_prefix): 

70 """ 

71 Get all terms that start with a specific prefix 

72  

73 Args: 

74 category_prefix: Prefix to match (e.g., "1." for all guidelines in section 1) 

75  

76 Returns: 

77 Dictionary of matching terms and explanations 

78 """ 

79 return {k: v for k, v in WCAG_GLOSSARY.items() if k.startswith(category_prefix)}

81# Function to get all WCAG guidelines 

82def get_all_wcag_guidelines(): 

83 """ 

84 Get all WCAG guidelines in the glossary 

85  

86 Returns: 

87 Dictionary of WCAG guidelines and explanations 

88 """ 

89 return {k: v for k, v in WCAG_GLOSSARY.items() if k[0].isdigit() and "." in k}
